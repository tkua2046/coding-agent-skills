"""Reviewer-only counterexamples for F1-F3; never edits canonical product files."""

import hashlib
import json
import subprocess
import sys
from pathlib import Path

snapshot = Path(sys.argv[1]).resolve()
evidence = Path(sys.argv[2]).resolve()
sys.path.insert(0, str(snapshot))
from tools.check import link_errors, validate_bundle

fixtures = snapshot.parent / "probes"
fixtures.mkdir()
results = []


def probe(label, location, content, expect_error):
    bundle = fixtures / label / "assets" / "example-skill"
    bundle.mkdir(parents=True)
    (bundle / "SKILL.md").write_text(
        "---\nname: example-skill\ndescription: Reviewer fixture.\n---\n# Skill\n"
    )
    (bundle / "rules.md").write_text("# Behavior\n")
    (bundle.parent / "outside.md").write_text("# Outside\n")
    target = bundle / location
    target.parent.mkdir(parents=True, exist_ok=True)
    text = content(bundle) if callable(content) else content
    target.write_text(text)
    errors = validate_bundle(bundle)
    results.append({
        "case": label, "file": str(target), "content": text,
        "expected_error": expect_error, "errors": errors,
        "matched_expectation": bool(errors) == expect_error,
    })


probe("asset-missing", "assets/design.template.md", "[rules](missing.md)\n", True)
probe("asset-escape", "assets/design.template.md", "[rules](../../outside.md)\n", True)
probe("asset-valid", "assets/design.template.md", "[rules](../rules.md#behavior)\n", False)
probe("missing-reference", "guide.md", "[rules][r]\n\n[r]: missing.md\n", True)
probe("reference-image", "guide.md", "![diagram][d]\n\n[d]: missing.png\n", True)
probe("linked-image", "guide.md", "[![diagram](missing.png)](https://example.com)\n", True)
probe("file-uri", "guide.md", "[rules](file:///nonexistent/rules.md)\n", True)
probe("absolute-in-bundle", "guide.md", lambda b: f"[rules]({b / 'rules.md'})\n", True)
probe("encoded-absolute-in-bundle", "guide.md", lambda b: "[rules](" + str(b / "rules.md").replace("/", "%2F") + ")\n", True)
probe("valid-reference-and-web", "guide.md", "[rules][r]\n\n[r]: rules.md#behavior\n\n[web](https://example.com)\n", False)
probe("inert-examples", "guide.md", "`[rules](missing.md)`\n\n```md\n[bad](file:///tmp/missing)\n```\n\n~~~md\n![x][m]\n\n[m]: missing.png\n~~~\n", False)

report_path = snapshot / "docs/reviews/implementation-initial.md"
report_errors = link_errors(report_path, snapshot)
results.append({"case": "unchanged-initial-report", "expected_error": False,
                "errors": report_errors, "matched_expectation": not report_errors})

legacy = snapshot / "docs/validation/hook-trials.json"
legacy.parent.mkdir(parents=True, exist_ok=True)
legacy.write_text('{"status":"legacy reviewer fixture"}\n')
legacy_hash = hashlib.sha256(legacy.read_bytes()).hexdigest()
old_reports = {}
for index in range(2):
    args = [sys.executable, "-m", "tests.manual.hook_trials"]
    result = subprocess.run(args, cwd=snapshot, text=True, capture_output=True, timeout=20)
    (evidence / f"harness-exception-{index + 1}.log").write_text(
        f"argv={args!r}\ncwd={snapshot}\nexit={result.returncode}\n"
        + result.stdout + result.stderr
    )
    current = sorted((snapshot / "docs/validation/hook-trials").glob("*.json"))
    retained = all(Path(name).read_bytes() == data for name, data in old_reports.items())
    for path in current:
        old_reports[str(path)] = path.read_bytes()
        destination = evidence / "harness-exception-reports" / path.name
        destination.parent.mkdir(exist_ok=True)
        if not destination.exists():
            destination.write_bytes(path.read_bytes())
    statuses = [json.loads(path.read_text())["status"] for path in current]
    preserved = hashlib.sha256(legacy.read_bytes()).hexdigest() == legacy_hash
    ok = result.returncode == 1 and len(current) == index + 1 and retained and preserved and all(s == "failed" for s in statuses)
    results.append({"case": f"harness-exception-{index + 1}", "exit": result.returncode,
                    "reason": "Sample directory deliberately absent from scoped copy",
                    "report_count": len(current), "statuses": statuses,
                    "prior_reports_unchanged": retained, "legacy_unchanged": preserved,
                    "matched_expectation": ok})

(evidence / "probe-results.json").write_text(json.dumps(results, indent=2) + "\n")
for result in results:
    print(json.dumps(result))
raise SystemExit(0 if all(r["matched_expectation"] for r in results) else 1)
