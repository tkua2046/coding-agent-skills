"""One-off scorer-only recovery of the two pinned v1 quote failures.

Default CLI operation validates only. --execute performs ONE fresh grading call.
The canonical engine, global grader and original reports are never edited.
"""

from __future__ import annotations

import argparse
import copy
import json
import sys
import tempfile
from pathlib import Path

ROOT = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(ROOT))
from tools import canary as c  # noqa: E402

ENGINE = "40df1bb684f719db6013129e741cce106b7176714d0b5ace9bc0a3eed80d4fe2"
GRADER = "17f84dbcc760a443f99076038e60faf52daa3c000f22d2b50f2f92d963244012"
CALIBRATION = "c510563e006ed5899863f80b7b7c1da58b3e100442b3fbf92a8e7322af3e98ca"
CONFIG = {"model": "gpt-6-astra", "effort": "medium", "timeout_seconds": 600}
SOURCES = {
    "20260907T084242-fd4d0ca28f534e23a8ef73b934ce8d97": "5ebfad94a8dfd0c290d0afc6e43fc4131587d9250ebb83a3d5182006701303ce",
    "20260907T084327-e8210bfcd7074d0581846925e88a1acf": "c56425fd2cb18785546b925d851606140e7db6ec273d41dfbed10c79a61d2dd0",
}
QUOTE_ERROR = "Grader cited missing evidence or an invented quote"


def require(condition, message):
    if not condition:
        raise ValueError(message)


def complete_probe(path):
    probe = c.read_json(path)
    require(
        probe.get("passed") is True
        and probe.get("exit_code") == 0
        and probe.get("timed_out") is False
        and probe.get("interrupted") is False
        and json.loads(probe["stdout"])
        == {
            "workspace_io": True,
            "private_read_denied": True,
            "network_denied": True,
        },
        "Isolation evidence incomplete",
    )


def verified_source(root: Path, source: Path, expected_sha: str):
    """Fail closed, without inventing a valid grade to bypass canonical checks."""
    require(source.is_relative_to(root), "Source must be inside this repository")
    row = c.verified_record(source)
    require(
        c.digest(source.read_bytes()) == expected_sha == SOURCES.get(row["run_id"]),
        "Source report SHA does not match the pinned original",
    )
    directory = source.parent
    retained = c.files(directory)
    require(
        c.hashes({p: b for p, b in retained.items() if p != "report.json"})
        == row["evidence"],
        "Source evidence inventory changed",
    )
    require(
        row.get("kind") == "behavior"
        and row.get("status") == "inconclusive"
        and row.get("error") == "ValueError: " + QUOTE_ERROR
        and not row.get("regrade")
        and row.get("case_id") == "v1-small-feature",
        "Only the original v1 invalid-quote outcomes qualify",
    )
    require(
        c.engine_identity(root) == row["identity"]["engine"] == ENGINE, "Engine changed"
    )
    require(
        c.grading_identity(root) == row["identity"]["grader"] == GRADER,
        "Grader changed",
    )
    require(
        row["settings"] == {"worker": CONFIG, "grader": CONFIG}, "Settings mismatch"
    )
    require(row["environment"] == c.environment(), "Environment mismatch")
    case = c.cases(root)[row["case_id"]]
    require(
        c.identity(root, case, row.get("source_ref")) == row["identity"],
        "Input identity mismatch",
    )
    require(
        c.fingerprint(c.hashes(c.files(directory / "inputs/case")))
        == row["identity"]["case"]
        and c.hashes(c.files(directory / "inputs/bundles"))
        == row["identity"]["skills"],
        "Archived inputs mismatch",
    )
    calibration = directory / "calibration/report.json"
    require(
        c.digest(calibration.read_bytes())
        == row["calibration"]["sha256"]
        == CALIBRATION,
        "Calibration link mismatch",
    )
    c.check_calibration(root, calibration, CONFIG, row["environment"])
    for name in ("isolation.json", "grader-isolation.json"):
        complete_probe(directory / name)
    # Deliberately bounded to these planning records: no oracle replay, hook,
    # overlay, conditional review or new worker execution is supported here.
    require(
        case["phases"] == [{"id": "01-plan", "request": "requests/01.md"}]
        and not case.get("hook")
        and not case.get("workflow_budget_seconds"),
        "Unsupported source workflow",
    )
    initial = c.files(directory / "initial")
    require(
        initial
        == {
            **c.files(directory / "inputs/case/fixture"),
            **c.files(directory / "inputs/bundles"),
        },
        "Initial fixture mismatch",
    )
    phase = directory / "phases/01-plan"
    worker = c.read_json(phase / "execution.json")
    c.require_completed(worker)
    require(
        (phase / "reply.md").read_text() == worker["reply"], "Worker reply mismatch"
    )
    request = (directory / "inputs/case/requests/01.md").read_bytes()
    require((phase / "request.md").read_bytes() == request, "Request mismatch")
    final = phase / "workspace"
    require(
        c.hashes({"skills/" + p: b for p, b in c.files(final / "skills").items()})
        == row["identity"]["skills"],
        "Worker changed supplied skills",
    )
    initial_git = c.read_json(directory / "initial-git.json")
    final_git = c.read_json(phase / "git.json")
    require(
        initial_git["head"] == row["fixture_head"] == final_git["head"],
        "Worker committed",
    )
    checks = []
    for check in case["checks"]:
        if check["kind"] == "unchanged":
            detail = {p: c.preserved(final, p, initial) for p in check["paths"]}
            require(all(detail.values()), "Preservation check failed")
        elif check["kind"] == "no_tags":
            require(final_git["tags"] == "", "Worker tagged")
            detail = {"tags": ""}
        elif check["kind"] == "commit_count":
            require(check["expected"] == 0, "Unsupported commit check")
            detail = {"added_commits": 0, "expected": 0}
        else:
            raise ValueError("Unsupported deterministic check")
        checks.append({"id": check["id"], "status": "pass", "evidence": detail})
    checks.append({"id": "runner.skills-preserved", "status": "pass"})
    require(
        checks == row["deterministic"] == c.read_json(directory / "deterministic.json"),
        "Deterministic evidence mismatch/failure",
    )
    observed = c.observations(directory, case)
    require(
        observed
        == row["observations"]
        == c.read_json(directory / "phases/observations.json")
        and not observed["unreached_phases"]
        and not observed["skipped_phases"],
        "Worker observations incomplete/mismatched",
    )
    reader = directory / "phases/reading-probe"
    exposed = c.reading_input(final, case["reading_probe"])
    require(
        c.read_json(reader / "input.json") == exposed
        and c.files(reader / "excerpts")
        == {
            f"{i}.txt": document["text"].encode()
            for i, document in enumerate(exposed["documents"])
        }
        and (reader / "prompt.md").read_text() == c.READER_PROMPT,
        "Reader input mismatch",
    )
    complete_probe(reader / "isolation.json")
    reading = c.read_json(reader / "execution.json")
    c.require_completed(reading)
    require(
        (reader / "reply.md").read_text() == reading["reply"], "Reader reply mismatch"
    )
    original = c.read_json(directory / "grade-reply.execution.json")
    c.require_completed(original)
    require(
        (directory / "grade-reply.json").read_text() == original["reply"],
        "Grader reply mismatch",
    )
    proposed = json.loads(original["reply"])
    rubric = c.read_json(directory / "inputs/case/rubric.json")
    criteria = {item["id"]: item for item in rubric["criteria"]}
    require(
        len(proposed["criteria"]) == len(criteria)
        and {item["id"] for item in proposed["criteria"]} == set(criteria),
        "Invalid criteria",
    )
    invalid_quotes = 0
    for item in proposed["criteria"]:
        require(
            item["status"] == "pass",
            "Do not retry semantic failures/inconclusive judgments",
        )
        try:
            c.validate_grade(
                {"criteria": [item]},
                {
                    "criteria": [
                        {**criteria[item["id"]], "required": True},
                    ]
                },
                directory,
                c.files(directory / "phases"),
            )
        except ValueError as exc:
            require(
                str(exc) == QUOTE_ERROR, "Original grading failed for another cause"
            )
            invalid_quotes += 1
    require(
        invalid_quotes > 0, "Original grading did not fail exact quotation validation"
    )
    return row, case, retained


def inherited_files(retained):
    return {
        p: b
        for p, b in retained.items()
        if p.startswith(("initial/", "inputs/", "phases/", "calibration/"))
        or p in {"initial-git.json", "isolation.json", "deterministic.json"}
    }


def verify_regrade(root: Path, path: Path):
    """Verify the additional reuse lineage, then canonical behavior acceptance."""
    row = c.verified_record(path)
    directory = path.parent
    link = row["regrade"]
    original = directory / "source/report.json"
    source_row = c.verified_record(original)
    require(
        c.hashes(
            {p: b for p, b in c.files(original.parent).items() if p != "report.json"}
        )
        == source_row["evidence"],
        "Regrade source inventory mismatch",
    )
    require(
        c.digest(original.read_bytes())
        == link["source_report_sha256"]
        == SOURCES.get(source_row["run_id"]),
        "Regrade source SHA mismatch",
    )
    require(
        source_row["run_id"] == link["source_run_id"]
        and link["source_report_path"]
        == c.read_json(directory / "attempt.json")["regrade"]["source_report_path"],
        "Regrade source link mismatch",
    )
    require(
        link["worker_reused"] is True
        and link["reader_reused"] is True
        and link["fresh_worker_calls"] == link["fresh_reader_calls"] == 0
        and link["fresh_grader_calls"] == 1,
        "Regrade misstates execution reuse",
    )
    for key in (
        "case_id",
        "identity",
        "settings",
        "environment",
        "source_ref",
        "calibration",
        "fixture_head",
        "observations",
        "deterministic",
    ):
        require(row[key] == source_row[key], "Regrade changed inherited " + key)
    require(
        all(
            (directory / p).read_bytes() == b
            for p, b in inherited_files(c.files(original.parent)).items()
        ),
        "Regrade changed inherited evidence",
    )
    require(
        c.digest((directory / "regrade-driver.py.txt").read_bytes())
        == link["helper_sha256"]
        == c.digest(Path(__file__).read_bytes()),
        "Regrade helper snapshot mismatch",
    )
    require(
        (directory / "regrade-prompt.md").read_bytes()
        == (root / "evals/graders/review.md").read_bytes(),
        "Regrade prompt mismatch",
    )
    require(
        link["new_grader_elapsed_seconds"]
        == c.read_json(directory / "grade-reply.execution.json")["elapsed_seconds"],
        "Regrade scorer timing mismatch",
    )
    return c.verified_behavior(root, path, c.cases(root)[row["case_id"]])


def regrade(root: Path, source: Path, expected_sha: str) -> Path:
    source_row, case, retained = verified_source(root, source, expected_sha)
    for attempt in (root / "docs/validation/canary").glob("*/attempt.json"):
        require(
            c.read_json(attempt).get("regrade", {}).get("source_report_sha256")
            != expected_sha,
            "A recovery attempt already exists; no automatic or implicit retry",
        )
    directory, row = c.new_record(root, "behavior")
    for key in (
        "case_id",
        "identity",
        "settings",
        "environment",
        "source_ref",
        "calibration",
        "fixture_head",
        "observations",
        "deterministic",
    ):
        row[key] = copy.deepcopy(source_row[key])
    helper = Path(__file__).read_bytes()
    row["regrade"] = {
        "operation": "one-off-grader-only-recovery",
        "source_report_path": str(source),
        "source_report_sha256": expected_sha,
        "source_run_id": source_row["run_id"],
        "worker_reused": True,
        "reader_reused": True,
        "fresh_worker_calls": 0,
        "fresh_reader_calls": 0,
        "fresh_grader_calls": 0,
        "worker_timing": "Original observations are reused, not new execution time.",
        "helper_sha256": c.digest(helper),
    }
    c.save(directory / "attempt.json", row)
    try:
        c.write_files(directory / "source", retained)
        inherited = inherited_files(retained)
        c.write_files(directory, inherited)
        c.write_files(directory, {"regrade-driver.py.txt": helper})
        prompt = (root / "evals/graders/review.md").read_bytes()
        c.write_files(directory, {"regrade-prompt.md": prompt})
        rubric = c.read_json(directory / "inputs/case/rubric.json")
        packet_files = {
            **{"artifacts/" + p: b for p, b in c.files(directory / "phases").items()},
            **{"initial/" + p: b for p, b in c.files(directory / "initial").items()},
            **{
                "requests/" + p["id"] + ".md": (
                    directory / "inputs/case" / p["request"]
                ).read_bytes()
                for p in case["phases"]
            },
        }
        # Canonical grade_packet/execute create these two files themselves.
        expected_packet = {
            **packet_files,
            "rubric.json": (json.dumps(rubric, indent=2) + "\n").encode(),
            "response-schema.json": json.dumps(c.grade_schema(rubric)).encode(),
        }
        c.save(directory / "regrade-packet-hashes.json", c.hashes(expected_packet))
        with tempfile.TemporaryDirectory(prefix="outcome-regrade-") as temp:
            packet = Path(temp).resolve()
            c.write_files(packet, packet_files)
            isolation = c.runtime.probe(
                packet, directory / "source/inputs/case/rubric.json"
            )
            c.save(directory / "grader-isolation.json", isolation)
            complete_probe(directory / "grader-isolation.json")
            verified_source(root, source, expected_sha)
            require(
                (root / "evals/graders/review.md").read_bytes() == prompt,
                "Prompt changed",
            )
            row["regrade"]["fresh_grader_calls"] = 1
            execution, grade, status = c.grade_packet(
                root,
                packet,
                directory / "grade-reply.json",
                rubric,
                CONFIG,
            )
            c.save(directory / "grade.json", grade)
            c.require_completed(execution)
            require(
                c.files(packet) == expected_packet, "Grader changed its evidence packet"
            )
            require(
                c.validate_grade(
                    grade, rubric, directory, c.files(directory / "phases")
                )
                == status,
                "Grading differs from immutable evidence",
            )
        require(
            c.files(source.parent) == retained,
            "Original source changed during recovery",
        )
        verified_source(root, source, expected_sha)
        require(
            c.files(directory / "source") == retained
            and all((directory / p).read_bytes() == b for p, b in inherited.items()),
            "Reused evidence changed",
        )
        row["semantic"] = row["status"] = status
        row["regrade"]["new_grader_elapsed_seconds"] = execution["elapsed_seconds"]
        # Validate a disposable final record before publishing a passing label.
        with tempfile.TemporaryDirectory(prefix="outcome-regrade-verify-") as temp:
            preview = Path(temp).resolve()
            c.write_files(preview, c.files(directory))
            preview_report = c.finish(preview, copy.deepcopy(row))
            verify_regrade(root, preview_report)
    except (Exception, KeyboardInterrupt) as exc:
        row["status"] = "inconclusive"
        row["error"] = f"{type(exc).__name__}: {exc}"
    execution_path = directory / "grade-reply.execution.json"
    if execution_path.is_file():
        row["regrade"]["new_grader_elapsed_seconds"] = c.read_json(execution_path).get(
            "elapsed_seconds"
        )
    return c.finish(directory, row)


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("source", type=Path)
    parser.add_argument("--source-sha256", required=True)
    parser.add_argument(
        "--execute", action="store_true", help="Make one real scoring call"
    )
    args = parser.parse_args()
    source = args.source.resolve()
    if args.execute:
        path = regrade(ROOT, source, args.source_sha256)
        print(path)
        return int(c.read_json(path)["status"] != "pass")
    verified_source(ROOT, source, args.source_sha256)
    print("Eligible pinned source; no model called, no recovery attempt created.")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
