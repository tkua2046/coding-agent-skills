import pytest

from counter import Counter


def test_default_add():
    counter = Counter()
    assert counter.add() == 1
    assert counter.add() == 2


@pytest.mark.parametrize("step", [1, 3, 10**100])
def test_positive_step(step):
    counter = Counter()

    assert counter.add(step) == step
    assert counter.value == step


def test_steps_accumulate_with_default_and_keyword_calls():
    counter = Counter()

    assert counter.add() == 1
    assert counter.add(step=3) == 4
    assert counter.add(2) == 6
    assert counter.add() == 7
    assert counter.value == 7


@pytest.mark.parametrize("initial_value", [0, 7])
@pytest.mark.parametrize("step", [0, -1, True, False, 1.0, 0.0, "3", None])
def test_invalid_step_does_not_mutate_counter(initial_value, step):
    counter = Counter()
    for _ in range(initial_value):
        counter.add()

    with pytest.raises(ValueError):
        counter.add(step)

    assert counter.value == initial_value
    assert counter.add() == initial_value + 1
