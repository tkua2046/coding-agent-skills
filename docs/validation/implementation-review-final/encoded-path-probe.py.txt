"""Three independent final-delta cases; operates only in the supplied snapshot."""
import json
import sys
from pathlib import Path

snapshot = Path(sys.argv[1]).resolve()
sys.path.insert(0, str(snapshot))
from tools.check import validate_bundle

records = []
for label in ["plain-absolute", "encoded-absolute", "encoded-relative"]:
    bundle = snapshot.parent / (snapshot.name + "-fixtures") / label / "example-skill"
    bundle.mkdir(parents=True)
    (bundle / "SKILL.md").write_text(
        "---\nname: example-skill\ndescription: Final review fixture.\n---\n# Skill\n"
    )
    support = bundle / "support rules.md"
    support.write_text("# Examples\n")
    if label == "encoded-relative":
        target = "support%20rules.md#examples"
    else:
        target = str(support).replace(" ", "%20")
        if label == "encoded-absolute":
            target = target.replace("/", "%2F")
    source = f"[rules]({target})\n"
    (bundle / "guide.md").write_text(source)
    errors = validate_bundle(bundle)
    ok = not errors if label == "encoded-relative" else any(
        "nonportable local resource" in error for error in errors
    )
    records.append({"case": label, "source": source, "existing_target": str(support),
                    "errors": errors, "matched_expected_result": ok})
print(json.dumps(records, indent=2))
raise SystemExit(0 if all(r["matched_expected_result"] for r in records) else 1)
